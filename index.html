🔗 Konfigurasi Auto-Sync Google Spreadsheet (Real-Time Webhook)

1. Buka Google Spreadsheet baru ➔ Extensions > Apps Script.
2. Salin kode `doPost(e)` di bawah, klik Deploy > New Deployment > Web App (Who has access: Anyone).
3. Tempel URL Web App di kolom bawah ini agar setiap jawaban murid otomatis masuk ke baris baru! 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu", "Kelas", "Nama Murid", "Kode Gedung", "Kegiatan", "Nomor & Pertanyaan", "Jawaban Murid", "Kunci Jawaban", "Status", "Poin", "Skor Total"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.timestamp, data.className, data.studentName, data.buildingCode, data.activityName, data.questionText, data.studentAnswer, data.correctAnswer, data.status, data.points, data.totalScore]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
